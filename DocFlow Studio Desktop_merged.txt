function cursorFromOffset(content: string, offset: number): { line: number; column: number } {
  const before = content.slice(0, Math.max(0, Math.min(offset, content.length)))
  const lines = before.split('\n')
  return { line: lines.length, column: (lines[lines.length - 1]?.length ?? 0) + 1 }
}

  const internalRef = useRef<HTMLTextAreaElement | null>(null)
  const gutterRef = useRef<HTMLDivElement>(null)
  const backdropRef = useRef<HTMLDivElement>(null)
  const popoverRef = useRef<HTMLDivElement>(null)

  const [scrollTop, setScrollTop] = useState(0)
  const [viewportHeight, setViewportHeight] = useState(600)
  const [currentLine, setCurrentLine] = useState(1)
  const [showBookmarkList, setShowBookmarkList] = useState(false)

  // Search & bookmark state (use scalar selectors to avoid object-reference churn)
  const searchQuery = useSearchStore((s) => s.query)
  const searchMatchCase = useSearchStore((s) => s.options.matchCase)
  const searchMode = useSearchStore((s) => s.options.mode)
  const searchWholeWord = useSearchStore((s) => s.options.matchWholeWord)
  const searchDotNewline = useSearchStore((s) => s.options.dotMatchesNewline)
  const isSearchOpen = useSearchStore((s) => s.isOpen)
  const lastMatchIndex = useSearchStore((s) => s.lastMatch?.index ?? null)
  const lastMatchLength = useSearchStore((s) => s.lastMatch?.length ?? null)
  const activeTabId = useTabStore((s) => s.activeTabId) ?? ''
  // Use a stable selector: return the stored array or undefined, then fall back
  // to a module-level constant to avoid a new [] reference every render.
  const bookmarkLineNums = useBookmarkStore((s) => s.bookmarks[activeTabId]) ?? EMPTY_LINES

  const {
    bookmarks,
    toggleBookmark,
    goToLine,
    goToNextBookmark,
    goToPrevBookmark,
    copyMarkedLines,
    clearMarks
  } = useBookmarks(internalRef)

  const setRefs = useCallback(
    (el: HTMLTextAreaElement | null) => {
      internalRef.current = el
      if (typeof ref === 'function') ref(el)
      else if (ref) (ref as React.MutableRefObject<HTMLTextAreaElement | null>).current = el
    },
    [ref]
  )

  // ── Resize observer ─────────────────────────────────────────────────────────
  useEffect(() => {
    if (!internalRef.current || typeof ResizeObserver === 'undefined') return
    const resizeObserver = new ResizeObserver((entries) => {
      for (const entry of entries) {
        setViewportHeight(entry.contentRect.height)
      }
    })
    resizeObserver.observe(internalRef.current)
    return () => resizeObserver.disconnect()
  }, [])

  // ── Popover outside-click ────────────────────────────────────────────────────
  useEffect(() => {
    if (!showBookmarkList) return
    const handler = (e: MouseEvent) => {
      if (popoverRef.current && !popoverRef.current.contains(e.target as Node)) {
        setShowBookmarkList(false)
      }
    }
    window.addEventListener('mousedown', handler)
    return () => window.removeEventListener('mousedown', handler)
  }, [showBookmarkList])

  // ── Cursor tracking ──────────────────────────────────────────────────────────
  const updateCursor = (element: HTMLTextAreaElement) => {
    const before = element.value.slice(0, element.selectionStart)
    const lines = before.split('\n')
    const lineNum = lines.length
    setCurrentLine(lineNum)
    onCursorChange({
      line: lineNum,
      column: (lines[lines.length - 1]?.length ?? 0) + 1,
      selection: Math.abs(element.selectionEnd - element.selectionStart)
    })
  }

  const handleKeyDown = (event: KeyboardEvent<HTMLTextAreaElement>) => {
    if (event.key !== 'Tab') return
    event.preventDefault()
    const element = event.currentTarget
    element.setRangeText('  ', element.selectionStart, element.selectionEnd, 'end')
    onChange(element.value)
    updateCursor(element)
  }

  const handleScroll = (event: UIEvent<HTMLTextAreaElement>) => {
    const target = event.currentTarget
    const st = target.scrollTop
    setScrollTop(st)
    if (gutterRef.current) gutterRef.current.scrollTop = st
    if (backdropRef.current) backdropRef.current.scrollTop = st
  }

  // ── Virtual gutter ───────────────────────────────────────────────────────────
  const allLines = value.split('\n')
  const lineCount = allLines.length
  const lineHeight = fontSize === 'sm' ? 21.875 : fontSize === 'lg' ? 28 : 24.5

  const visibleStart = Math.max(1, Math.floor(scrollTop / lineHeight) - 5)
  const visibleEnd = Math.min(lineCount, Math.ceil((scrollTop + viewportHeight) / lineHeight) + 10)

  const topSpacer = Math.max(0, (visibleStart - 1) * lineHeight)
  const bottomSpacer = Math.max(0, (lineCount - visibleEnd) * lineHeight)

  const visibleLines: number[] = []
  for (let i = visibleStart; i <= visibleEnd; i++) {
    visibleLines.push(i)
  }

  // ── Highlight segments (memoised) ────────────────────────────────────────────
  const highlightSegments = useMemo(() => {
    const allRanges = []

    // 1. Bookmark-line ranges
    const bmRanges = bookmarkLinesToRanges(value, bookmarkLineNums)
    allRanges.push(...bmRanges)

    // 2. Search match ranges (only when search bar is open and query exists)
    if (isSearchOpen && searchQuery) {
      try {
        const regex = buildRegExp(searchQuery, {
          matchCase: searchMatchCase,
          mode: searchMode,
          matchWholeWord: searchWholeWord,
          dotMatchesNewline: searchDotNewline,
          inSelection: false
        })
        const matches = findAll(value, regex)
        const currentIdx = lastMatchIndex
        const currentLen = lastMatchLength
        const searchRanges = searchMatchesToRanges(
          matches.map((m) => ({ index: m.index, length: m.length })),
          currentIdx,
          currentLen
        )
        allRanges.push(...searchRanges)
      } catch {
        // Invalid regex — no search highlights
      }
    }

    if (allRanges.length === 0) return null
    return buildSegments(value, allRanges)
  }, [value, bookmarkLineNums, isSearchOpen, searchQuery, searchMatchCase, searchMode, searchWholeWord, searchDotNewline, lastMatchIndex, lastMatchLength])

  const showBackdrop = highlightSegments !== null && highlightSegments.length > 0

  // ── Render ───────────────────────────────────────────────────────────────────
  return (
    <section className="editor-pane">
      <div className="pane-header">
        <div className="pane-header-left">
          <span className="pane-label">Markdown</span>
          <span className="pane-hint">{lineCount} lines</span>
        </div>

        <div className="bookmark-toolbar">
          <button
            type="button"
            className="icon-button"
            title={t('find:toggleBookmark')}
            onClick={() => toggleBookmark(currentLine)}
          >
            <Bookmark
              size={13}
              fill={bookmarks.includes(currentLine) ? 'currentColor' : 'none'}
              color={bookmarks.includes(currentLine) ? 'var(--accent)' : 'currentColor'}
            />
          </button>

          {bookmarks.length > 0 && (
            <>
              <button
                type="button"
                className="bookmark-badge-btn"
                title={`${bookmarks.length} ${t('find:bookmarks')} — Click to view list`}
                onClick={() => setShowBookmarkList((v) => !v)}
              >
                <Bookmark size={11} fill="currentColor" /> {bookmarks.length}
              </button>

              <button
                type="button"
                className="icon-button"
                title={t('find:prevBookmark')}
                onClick={() => goToPrevBookmark()}
              >
                <ChevronUp size={13} />
              </button>

              <button
                type="button"
                className="icon-button"
                title={t('find:nextBookmark')}
                onClick={() => goToNextBookmark()}
              >
                <ChevronDown size={13} />
              </button>

              <button
                type="button"
                className="icon-button"
                title={t('find:copyMarked')}
                onClick={() => void copyMarkedLines()}
              >
                <Copy size={13} />
              </button>

              <button
                type="button"
                className="icon-button"
                title={t('find:clearAllBookmarks')}
                onClick={() => clearMarks()}
              >
                <BookmarkX size={13} />
              </button>
            </>
          )}

          {showBookmarkList && (
            <div className="bookmark-popover" ref={popoverRef}>
              <div className="bookmark-popover-head">
                <strong>
                  {t('find:bookmarks')} ({bookmarks.length})
                </strong>
                <button
                  type="button"
                  className="icon-button"
                  onClick={() => clearMarks()}

                  <BookmarkX size={13} />
                </button>
              </div>
              <div className="bookmark-popover-list">
                {bookmarks.map((lineNum) => {

                    <button
                      key={lineNum}
                      type="button"
                      className="bookmark-popover-item"
                      onClick={() => {

                        setShowBookmarkList(false)
                      }}
                    >

                        {lineContent.trim() || '(empty line)'}

                    </button>
                  )
                })}
              </div>
            </div>
          )}

      </div>

import { forwardRef, type MouseEvent, type UIEvent } from 'react'
import type { DocumentSettings } from '../types/document'
import { copyText } from '../utils/clipboard'
import { useFileStore } from '../store/fileStore'
import { describeError } from '../utils/errors'

interface DocumentViewerProps {
  html: string
  title: string
  settings: DocumentSettings
  zen?: boolean
  zoomLevel?: number
  isPagedMode?: boolean
  onScrollTop?: (top: number) => void
}

export const DocumentViewer = forwardRef<HTMLDivElement, DocumentViewerProps>(function DocumentViewer(
  { html, title, settings, zen = false, zoomLevel = 100, isPagedMode = true, onScrollTop },
  ref
) {
  const { handleContextMenu } = useContextMenu()

  const handleClick = async (event: MouseEvent<HTMLDivElement>) => {
    const target = event.target as HTMLElement
    const copyButton = target.closest<HTMLButtonElement>('[data-copy-code]')
    if (copyButton) {
      const block = copyButton.closest('.code-block')
      const code = block?.querySelector('code')?.textContent ?? ''
      const ok = await copyText(code)
      copyButton.textContent = ok ? 'Copied' : 'Failed'
      window.setTimeout(() => {
        copyButton.textContent = 'Copy'
      }, 1400)
      return
    }

    const anchor = target.closest<HTMLAnchorElement>('a[href]')
    if (!anchor) return
    const href = anchor.getAttribute('href') ?? ''
    if (href.startsWith('#')) {
      event.preventDefault()
      const destination = document.getElementById(href.slice(1))
      destination?.scrollIntoView({ behavior: 'smooth', block: 'start' })
      return
    }
    if (/^https?:\/\//i.test(href)) {
      event.preventDefault()
      try {
        await window.docflow.app.openExternal(href)
      } catch (error) {
        const described = describeError(error)
        useFileStore.getState().showToast(described.kind, described.message, described.detail)
      }
    }
  }

  const today = new Date().toLocaleDateString(undefined, {
    year: 'numeric',
    month: 'short',
    day: 'numeric'
  })

  return (
    <section
      ref={ref}
      className={`preview-pane ${zen ? 'zen-preview' : ''}`}
      onContextMenu={(e) => handleContextMenu(e, 'preview')}
      onScroll={(event: UIEvent<HTMLElement>) => onScrollTop?.(event.currentTarget.scrollTop)}
    >
      <div className="preview-stage">
        <article
          style={{ transform: `scale(${zoomLevel / 100})`, transformOrigin: 'top left' }}
          className={`document-page theme-${settings.theme} paper-${settings.paperTone} font-${settings.fontFamily} size-${settings.fontSize} margin-${settings.marginSize} ${settings.justifyText ? 'justify-text' : ''} ${settings.autoNumberHeadings ? 'numbered-headings' : ''} ${isPagedMode ? 'paged-mode' : 'continuous-mode'}`}
        >
          {settings.showHeaderFooter && (
            <header className="doc-header">
              <span className="doc-header-title">{title}</span>
              <span className="doc-header-meta">DocFlow Studio</span>
            </header>
          )}
          <div
            className="markdown-body"
            onClick={(event) => void handleClick(event)}
            dangerouslySetInnerHTML={{ __html: html }}
          />
          {settings.showHeaderFooter && (
            <footer className="doc-footer">
              <span>{today}</span>
              {settings.showPageNumbers && <span className="doc-page-number">1</span>}
            </footer>
          )}
        </article>
      </div>
    </section>
  )
})

import { useCallback } from 'react'
import type { RefObject } from 'react'
import type { FindOptions, FindResult } from '../../electron/types/ipc'
import { useTabStore } from '../store/tabStore'
import { useSearchStore, FIND_RESULTS_MAX } from '../store/searchStore'
import { useFileStore } from '../store/fileStore'
import { useI18n } from './useI18n'
import {
  buildRegExp,
  decodeExtendedEscapes,
  findAll,
  lineColumnAt,
  lineTextAt
} from '../utils/searchRegex'
import { describeError } from '../utils/errors'

interface MatchRange {
  index: number
  length: number
}

function selectionBounds(content: string, options: FindOptions): { start: number; end: number } {
  if (!options.inSelection) return { start: 0, end: content.length }
  const range = useSearchStore.getState().selectionRange
  if (!range || range.end <= range.start) return { start: 0, end: content.length }
  return { start: Math.max(0, range.start), end: Math.min(content.length, range.end) }
}

function computeReplacement(match: RegExpExecArray, replacement: string, mode: FindOptions['mode']): string {
  if (mode !== 'regex') return replacement
  return match[0].replace(/.*/, () =>
    replacement.replace(/\$(\$|&|\d{1,2})/g, (token, group: string) => {
      if (group === '$') return '$'
      if (group === '&') return match[0]
      const index = Number.parseInt(group, 10)
      return match[index] ?? token
    })
  )
}

function replacementText(options: FindOptions, replaceWith: string): string {
  return options.mode === 'extended' ? decodeExtendedEscapes(replaceWith, options.extendedUnicode) : replaceWith
}

export interface FindReplaceApi {
  findNext(): boolean
  findPrevious(): boolean
  replaceOne(): boolean
  replaceAll(): number
  countMatches(): number
  findAllInTabs(): number
  goToResult(result: FindResult): void
  clearSelectionRange(): void
}

export function useFindReplace(editorRef: RefObject<HTMLTextAreaElement | null>): FindReplaceApi {
  const { t } = useI18n()

  const selectMatch = useCallback(
    (tabId: string, content: string, match: MatchRange, focusEditor = false) => {
      const store = useSearchStore.getState()
      store.setLastMatch({ tabId, index: match.index, length: match.length })
      const end = match.index + match.length
      useTabStore.getState().updateTab(tabId, { cursorOffset: match.index, selectionEnd: end })
      const editor = editorRef.current
      if (editor) {
        if (focusEditor) editor.focus()
        editor.setSelectionRange(match.index, end)
        const before = content.slice(0, match.index)
        const lineIndex = before.split('\n').length - 1
        const lineHeight = Number.parseFloat(window.getComputedStyle(editor).lineHeight) || 20
        const top = Math.max(0, lineIndex * lineHeight - editor.clientHeight / 3)
        editor.scrollTo({ top, behavior: 'smooth' })
      }
    },
    [editorRef]
  )

  const collectMatches = useCallback((content: string, options: FindOptions): MatchRange[] => {
    const { start, end } = selectionBounds(content, options)
    const scoped = content.slice(start, end)
    const matches = findAll(scoped, buildRegExp(useSearchStore.getState().query, options))
    return matches.map((match) => ({ index: match.index + start, length: match.length }))
  }, [])

  const findNext = useCallback((): boolean => {
    const state = useSearchStore.getState()
    const tab = useTabStore.getState().getActive()
    if (!tab) return false
    try {
      if (!state.query) return false
      const matches = collectMatches(tab.fileContent, state.options)
      if (matches.length === 0) {
        state.setLastMatch(null)
        return false
      }
      const editor = editorRef.current
      const activeEditor = document.activeElement === editor
      const last = state.lastMatch?.tabId === tab.id ? state.lastMatch : null
      const anchor = last ? last.index + last.length : activeEditor ? (editor?.selectionEnd ?? 0) : 0
      let target = matches.find((match) => match.index >= anchor)
      if (!target && state.options.wrapAround) target = matches[0]
      if (!target) return false
      const current = target
      if (last && current.index === last.index && last.length === current.length) {
        const alternative = matches.find((match) => match.index > current.index)
        if (alternative) target = alternative
        else if (state.options.wrapAround) target = matches[0]
      }
      if (!target) return false
      selectMatch(tab.id, tab.fileContent, target)
      state.addHistory(state.query)
      return true
    } catch (error) {
      const described = describeError(error)
      useFileStore.getState().showToast('error', t('find:invalidRegex'), described.detail || described.message)
      return false
    }
  }, [collectMatches, editorRef, selectMatch, t])

  const findPrevious = useCallback((): boolean => {
    const state = useSearchStore.getState()
    const tab = useTabStore.getState().getActive()
    if (!tab) return false
    try {
      if (!state.query) return false
      const matches = collectMatches(tab.fileContent, state.options)
      if (matches.length === 0) {
        state.setLastMatch(null)
        return false
      }
      const editor = editorRef.current
      const last = state.lastMatch?.tabId === tab.id ? state.lastMatch : null
      const anchor = last ? last.index : (editor?.selectionStart ?? tab.fileContent.length)
      let target = [...matches].reverse().find((match) => match.index < anchor)
      if (!target && state.options.wrapAround) target = matches[matches.length - 1]
      if (!target) return false
      selectMatch(tab.id, tab.fileContent, target)
      state.addHistory(state.query)
      return true
    } catch (error) {
      const described = describeError(error)
      useFileStore.getState().showToast('error', t('find:invalidRegex'), described.detail || described.message)
      return false
    }
  }, [collectMatches, editorRef, selectMatch, t])

  const replaceOne = useCallback((): boolean => {
    const state = useSearchStore.getState()
    const tab = useTabStore.getState().getActive()
    const match = state.lastMatch
    if (!tab || !match || match.tabId !== tab.id) return findNext()
    try {
      const options = state.options
      const regex = buildRegExp(state.query, { ...options, inSelection: false })
      regex.lastIndex = match.index
      const found = regex.exec(tab.fileContent)
      if (!found || found.index !== match.index) return findNext()
      const replacement = replacementText(options, state.replaceWith)
      const value = computeReplacement(found, replacement, options.mode)
      const nextContent = `${tab.fileContent.slice(0, match.index)}${value}${tab.fileContent.slice(match.index + match.length)}`
      const editor = editorRef.current
      if (editor && document.activeElement === editor) {
        editor.setRangeText(value, match.index, match.index + match.length, 'end')
        useTabStore.getState().setBuffer(tab.id, editor.value)
      } else {
        useTabStore.getState().setBuffer(tab.id, nextContent)
        editor?.setSelectionRange(match.index + value.length, match.index + value.length)
      }
      useSearchStore.getState().setLastMatch(null)
      state.addHistory(state.query)
      return findNext()
    } catch (error) {
      const described = describeError(error)
      useFileStore.getState().showToast('error', t('find:invalidRegex'), described.detail || described.message)
      return false
    }
  }, [editorRef, findNext, t])

  const replaceAll = useCallback((): number => {
    const state = useSearchStore.getState()
    const tab = useTabStore.getState().getActive()
    if (!tab) return 0
    try {
      const options = state.options
      const regex = buildRegExp(state.query, options)
      const bounds = selectionBounds(tab.fileContent, options)
      const scoped = tab.fileContent.slice(bounds.start, bounds.end)
      const matches = findAll(scoped, regex)
      if (matches.length === 0) return 0
      const literal = replacementText(options, state.replaceWith)
      const replacement = options.mode === 'regex' ? state.replaceWith : literal
      const replaced = options.mode === 'regex'
        ? scoped.replace(regex, replacement)
        : scoped.replace(regex, () => literal)
      const content = `${tab.fileContent.slice(0, bounds.start)}${replaced}${tab.fileContent.slice(bounds.end)}`
      useTabStore.getState().setBuffer(tab.id, content)
      useSearchStore.getState().setLastMatch(null)
      state.addHistory(state.query)
      const editor = editorRef.current
      if (editor) {
        editor.setSelectionRange(bounds.start, bounds.start + replaced.length)
        editor.focus()
      }
      return matches.length
    } catch (error) {
      const described = describeError(error)
      useFileStore.getState().showToast('error', t('find:invalidRegex'), described.detail || described.message)
      return 0
    }
  }, [editorRef, t])

  const countMatches = useCallback((): number => {
    const state = useSearchStore.getState()
    const tab = useTabStore.getState().getActive()
    if (!tab || !state.query) return 0
    try {
      return collectMatches(tab.fileContent, state.options).length
    } catch {
      return 0
    }
  }, [collectMatches])

  const findAllInTabs = useCallback((): number => {
    const state = useSearchStore.getState()
    const tabs = useTabStore.getState().tabs
    try {
      const regexSource = buildRegExp(state.query, { ...state.options, inSelection: false })
      const results: FindResult[] = []
      let truncated = false
      for (const tab of tabs) {
        const regex = new RegExp(regexSource.source, regexSource.flags)
        const matches = findAll(tab.fileContent, regex)
        for (const match of matches) {
          if (results.length >= FIND_RESULTS_MAX) {
            truncated = true
            break
          }
          const { lineNumber, columnStart } = lineColumnAt(tab.fileContent, match.index)
          results.push({
            tabId: tab.id,
            filePath: tab.filePath ?? tab.displayName,
            lineNumber,
            columnStart,
            columnEnd: columnStart + match.length,
            preview: lineTextAt(tab.fileContent, match.index)
          })
        }
        if (truncated) break
      }
      state.addHistory(state.query)
      state.setAllResults(results, truncated)
      return results.length
    } catch (error) {
      const described = describeError(error)
      useFileStore.getState().showToast('error', t('find:invalidRegex'), described.detail || described.message)
      return 0
    }
  }, [t])

  const goToResult = useCallback(
    (result: FindResult) => {
      const tabStore = useTabStore.getState()
      const tab =
        (result.tabId ? tabStore.tabs.find((item) => item.id === result.tabId) : undefined) ??
        tabStore.tabs.find((item) => item.filePath === result.filePath)
      if (!tab) return
      tabStore.activateTab(tab.id)
      const lines = tab.fileContent.split('\n')
      const lineStart = lines.slice(0, Math.max(0, result.lineNumber - 1)).reduce((total, line) => total + line.length + 1, 0)
      const start = lineStart + Math.max(0, result.columnStart - 1)
      const end = start + Math.max(0, result.columnEnd - result.columnStart)
      useSearchStore.getState().setLastMatch({ tabId: tab.id, index: start, length: end - start })
      window.setTimeout(() => {
        const editor = editorRef.current
        if (!editor) return
        editor.focus()
        editor.setSelectionRange(start, end)
        const lineHeight = Number.parseFloat(window.getComputedStyle(editor).lineHeight) || 20
        editor.scrollTo({ top: Math.max(0, (result.lineNumber - 1) * lineHeight - editor.clientHeight / 3), behavior: 'smooth' })
      }, 30)
    },
    [editorRef]
  )

  const clearSelectionRange = useCallback(() => {
    useSearchStore.getState().setSelectionRange(null)
  }, [])

  return { findNext, findPrevious, replaceOne, replaceAll, countMatches, findAllInTabs, goToResult, clearSelectionRange }
}

import { useCallback, useRef } from 'react'
import { useUIStore } from '../store/uiStore'
import type { HeadingItem } from '../types/document'

export function useScrollSync(editorRef: React.RefObject<HTMLTextAreaElement | null>, previewRef: React.RefObject<HTMLDivElement | null>) {
  const syncScrollWithToc = useUIStore((s) => s.syncScrollWithToc)
  const autoScroll = useUIStore((s) => s.autoScroll)

  const handleTocClick = useCallback((item: HeadingItem) => {
    if (!syncScrollWithToc) return
    const previewEl = previewRef.current
    if (previewEl) {
      const target = previewEl.querySelector(`[id="${item.id}"]`) as HTMLElement | null
      if (target) {
        target.scrollIntoView({ behavior: 'smooth', block: 'start' })
      }
    }
    const editorEl = editorRef.current
    if (editorEl) {
      const text = editorEl.value
      const lines = text.split('\n')
      // Find first line matching heading level and text approx
      const levelPrefix = '#'.repeat(item.level)
      let lineIndex = -1
      for (let i = 0; i < lines.length; i++) {
        const trimmed = lines[i].trim()
        if (trimmed.startsWith(levelPrefix + ' ') && trimmed.includes(item.text)) {
          lineIndex = i
          break
        }
      }
      if (lineIndex >= 0) {
        const lineHeight = parseFloat(window.getComputedStyle(editorEl).lineHeight) || 20
        const scrollTop = Math.max(0, lineIndex * lineHeight - 40)
        editorEl.scrollTo({ top: scrollTop, behavior: 'smooth' })
        const pos = text.split('\n').slice(0, lineIndex).join('\n').length + (lineIndex > 0 ? 1 : 0)
        // Place cursor roughly at start of line
        editorEl.setSelectionRange(pos, pos)
        editorEl.focus()
      }
    }
  }, [syncScrollWithToc, editorRef, previewRef])

  const handleEditorScroll = useCallback(() => {
    if (!autoScroll) return
    const editorEl = editorRef.current
    const previewEl = previewRef.current
    if (!editorEl || !previewEl) return
    const ratio = editorEl.scrollTop / Math.max(1, editorEl.scrollHeight - editorEl.clientHeight)
    const maxScroll = Math.max(1, previewEl.scrollHeight - previewEl.clientHeight)
    previewEl.scrollTop = ratio * maxScroll
  }, [autoScroll, editorRef, previewRef])

  return { handleTocClick, handleEditorScroll }
}

import { create } from 'zustand'
import type { FilePayload, TabSnapshot, TextEncoding, LineEnding } from '../../electron/types/ipc'

interface TabStore {
  tabs: DocumentTab[]
  activeTabId: string | null

let idCounter = 0

function createId(): string {
  idCounter += 1
  const cryptoApi = typeof crypto !== 'undefined' ? crypto : null
  if (cryptoApi && typeof cryptoApi.randomUUID === 'function') return cryptoApi.randomUUID()
  return `tab-${Date.now()}-${idCounter}`
}

export function isTabDirty(tab: Pick<DocumentTab, 'fileContent' | 'savedContent'>): boolean {
  return tab.fileContent !== tab.savedContent
}

export function snapshotToTab(snapshot: TabSnapshot): DocumentTab {
  return {
    ...snapshot,
    conflict: null,
    externalChange: null
  }
}

function makeTab(options: {
  filePath: string | null
  displayName: string
  fileContent: string
  savedContent?: string
  diskMtime?: number | null
  encoding?: TextEncoding
  lineEnding?: LineEnding
  isPinned?: boolean
}): DocumentTab {
  return {
    id: createId(),
    filePath: options.filePath,
    displayName: options.displayName,
    fileContent: options.fileContent,
    savedContent: options.savedContent ?? options.fileContent,
    diskMtime: options.diskMtime ?? null,
    encoding: options.encoding ?? 'utf-8',
    lineEnding: options.lineEnding ?? 'lf',
    isPinned: options.isPinned ?? false,
    editorScrollTop: 0,
    previewScrollTop: 0,
    cursorOffset: 0,
    selectionEnd: 0,
    conflict: null,
    externalChange: null
  }
}

function sortPinned(tabs: DocumentTab[]): DocumentTab[] {
  const pinned = tabs.filter((tab) => tab.isPinned)
  const regular = tabs.filter((tab) => !tab.isPinned)
  return [...pinned, ...regular]
}

function neighbourId(tabs: DocumentTab[], id: string): string | null {
  const index = tabs.findIndex((tab) => tab.id === id)
  if (index < 0) return tabs[0]?.id ?? null
  return tabs[index + 1]?.id ?? tabs[index - 1]?.id ?? null
}

function initialStore() {
  const welcome = makeTab({
    filePath: null,
    displayName: 'Welcome.md',
    fileContent: WELCOME_DOCUMENT
  })
  return {
    tabs: [welcome],
    activeTabId: welcome.id,
    windowId: null,
    reopenStack: [] as TabSnapshot[],
    hydrated: false,
    untitledCounter: 1
  }
}

export const useTabStore = create<TabStore>((set, get) => ({
  ...initialStore(),

  hydrate: (tabs, activeTabId, windowId) => {
    const hydratedTabs = tabs.length > 0 ? tabs : initialStore().tabs
    const active = hydratedTabs.some((tab) => tab.id === activeTabId) ? activeTabId : hydratedTabs[0].id
    set({
      tabs: sortPinned(hydratedTabs),
      activeTabId: active,
      windowId,
      hydrated: true
    })
  },

  setWindowId: (windowId) => set({ windowId }),

  openTab: (payload, activate = true) => {
    const state = get()
    if (payload.path) {
      const existing = state.tabs.find((tab) => tab.filePath === payload.path)
      if (existing) {
        if (activate) set({ activeTabId: existing.id })
        return { ok: true, id: existing.id, reason: 'duplicate' }
      }
    }
    if (state.tabs.length >= MAX_TABS) return { ok: false, reason: 'max' }
    const tab = makeTab({
      filePath: payload.path,
      displayName: payload.name,
      fileContent: payload.content,
      savedContent: payload.content,
      diskMtime: payload.mtime,
      encoding: payload.encoding,
      lineEnding: payload.lineEnding
    })
    set({
      tabs: [...state.tabs, tab],
      activeTabId: activate ? tab.id : state.activeTabId
    })
    return { ok: true, id: tab.id }
  },

  openSnapshot: (snapshot, activate = true) => {
    const state = get()
    if (snapshot.filePath) {
      const existing = state.tabs.find((tab) => tab.filePath === snapshot.filePath)
      if (existing) {
        if (activate) set({ activeTabId: existing.id })
        return { ok: true, id: existing.id, reason: 'duplicate' }
      }
    }
    if (state.tabs.length >= MAX_TABS) return { ok: false, reason: 'max' }
    const tab = snapshotToTab({ ...snapshot, id: createId() })
    set({
      tabs: sortPinned([...state.tabs, tab]),
      activeTabId: activate ? tab.id : state.activeTabId
    })
    return { ok: true, id: tab.id }
  },

  newTab: (content = '', activate = true) => {
    const state = get()
    if (state.tabs.length >= MAX_TABS) return { ok: false, reason: 'max' }
    const name = `Untitled${state.untitledCounter > 1 ? ` ${state.untitledCounter}` : ''}.md`
    const tab = makeTab({
      filePath: null,
      displayName: name,
      fileContent: content
    })
    set({
      tabs: [...state.tabs, tab],
      activeTabId: activate ? tab.id : state.activeTabId,
      untitledCounter: state.untitledCounter + 1
    })
    return { ok: true, id: tab.id }
  },

  activateTab: (id) => {
    if (get().tabs.some((tab) => tab.id === id)) set({ activeTabId: id })
  },

  setBuffer: (id, content) =>
    set((state) => ({
      tabs: state.tabs.map((tab) => (tab.id === id ? { ...tab, fileContent: content } : tab))
    })),

  updateTab: (id, patch) =>
    set((state) => ({
      tabs: sortPinned(state.tabs.map((tab) => (tab.id === id ? { ...tab, ...patch, id } : tab)))
    })),

  closeTab: (id) => {
    const state = get()
    const target = state.tabs.find((tab) => tab.id === id)
    if (!target) return { closed: false, reason: 'missing' }
    const remaining = state.tabs.filter((tab) => tab.id !== id)
    const nextActive = state.activeTabId === id ? neighbourId(state.tabs, id) : state.activeTabId
    const reopen = remaining.length > 0 ? [...state.reopenStack, target].slice(-10) : state.reopenStack
    set({
      tabs: remaining,
      activeTabId: nextActive && remaining.some((tab) => tab.id === nextActive) ? nextActive : remaining[0]?.id ?? null,
      reopenStack: reopen
    })
    if (remaining.length === 0) {
      const store = get()
      store.newTab('', true)
    }
    return { closed: true }
  },

  closeOthers: (id) => {
    const state = get()
    const kept = state.tabs.filter((tab) => tab.id === id || tab.isPinned)
    set({
      tabs: kept,
      activeTabId: kept.some((tab) => tab.id === state.activeTabId) ? state.activeTabId : id
    })
  },

  closeToRight: (id) => {
    const state = get()
    const index = state.tabs.findIndex((tab) => tab.id === id)
    if (index < 0) return
    const kept = state.tabs.filter((tab, position) => position <= index || tab.isPinned)
    set({
      tabs: kept,
      activeTabId: kept.some((tab) => tab.id === state.activeTabId) ? state.activeTabId : id
    })
  },

  closeToLeft: (id) => {
    const state = get()
    const index = state.tabs.findIndex((tab) => tab.id === id)
    if (index < 0) return
    const kept = state.tabs.filter((tab, position) => position >= index || tab.isPinned)
    set({
      tabs: kept,
      activeTabId: kept.some((tab) => tab.id === state.activeTabId) ? state.activeTabId : id
    })
  },

  closeAll: () => {
    const state = get()
    const kept = state.tabs.filter((tab) => tab.isPinned)
    if (kept.length === 0) {
      const store = get()
      store.newTab('', true)
    } else {
      set({ tabs: kept, activeTabId: kept[0].id })
    }
  },

  togglePin: (id) =>
    set((state) => ({
      tabs: sortPinned(
        state.tabs.map((tab) => (tab.id === id ? { ...tab, isPinned: !tab.isPinned } : tab))
      )
    })),

  moveTab: (activeId, overId) => {
    const state = get()
    const from = state.tabs.findIndex((tab) => tab.id === activeId)
    const to = state.tabs.findIndex((tab) => tab.id === overId)
    if (from < 0 || to < 0 || from === to) return
    const source = state.tabs[from]
    const target = state.tabs[to]
    if (source.isPinned !== target.isPinned) return
    const tabs = [...state.tabs]
    const [moved] = tabs.splice(from, 1)
    tabs.splice(to, 0, moved)
    set({ tabs })
  },

  markSaved: (id, path, content, mtime) =>
    set((state) => ({
      tabs: state.tabs.map((tab) =>
        tab.id === id
          ? {
              ...tab,
              filePath: path,
              displayName: path.split(/[\\/]/).pop() || tab.displayName,
              fileContent: content,
              savedContent: content,
              diskMtime: mtime,
              conflict: null,
              externalChange: null
            }
          : tab
      )
    })),

  markSavedContent: (id, savedContent) =>
    set((state) => ({
      tabs: state.tabs.map((tab) => (tab.id === id ? { ...tab, savedContent } : tab))
    })),

  setConflict: (id, conflict) =>
    set((state) => ({
      tabs: state.tabs.map((tab) => (tab.id === id ? { ...tab, conflict } : tab))
    })),

  setExternalChange: (id, change) =>
    set((state) => ({
      tabs: state.tabs.map((tab) => (tab.id === id ? { ...tab, externalChange: change } : tab))
    })),

  setAllExternalChanges: (change) =>
    set((state) => ({
      tabs: state.tabs.map((tab) =>
        change && tab.filePath === change.path ? { ...tab, externalChange: change } : { ...tab, externalChange: null }
      )
    })),

  getActive: () => {
    const state = get()
    return state.tabs.find((tab) => tab.id === state.activeTabId) ?? null
  },

  getTab: (id) => get().tabs.find((tab) => tab.id === id) ?? null,

  toSnapshots: () =>
    get().tabs.map((tab) => ({
      id: tab.id,
      filePath: tab.filePath,
      displayName: tab.displayName,
      fileContent: tab.fileContent,
      savedContent: tab.savedContent,
      diskMtime: tab.diskMtime,
      encoding: tab.encoding,
      lineEnding: tab.lineEnding,
      isPinned: tab.isPinned,
      editorScrollTop: tab.editorScrollTop,
      previewScrollTop: tab.previewScrollTop,
      cursorOffset: tab.cursorOffset,
      selectionEnd: tab.selectionEnd
    })),

  pruneEmpty: () => {
    const state = get()
    const tabs = state.tabs.filter(
      (tab) => Boolean(tab.filePath) || tab.fileContent.trim().length > 0 || tab.savedContent.trim().length > 0
    )
    if (tabs.length === state.tabs.length) return
    if (tabs.length === 0) {
      const store = get()
      store.newTab('', true)
      return
    }
    set({
      tabs,
      activeTabId: tabs.some((tab) => tab.id === state.activeTabId) ? state.activeTabId : tabs[0].id
    })
  },

  setHydrated: (value) => set({ hydrated: value })
}))

export function useActiveTab(): DocumentTab | null {
  return useTabStore((state) => state.tabs.find((tab) => tab.id === state.activeTabId) ?? null)
}

export function useTabDirty(): boolean {
  return useTabStore((state) => {
    const active = state.tabs.find((tab) => tab.id === state.activeTabId)
    return active ? active.fileContent !== active.savedContent : false
  })
}
